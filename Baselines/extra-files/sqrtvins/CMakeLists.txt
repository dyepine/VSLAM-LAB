cmake_minimum_required(VERSION 3.16)
project(vslamlab_sqrtvins LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system thread date_time)
file(GLOB_RECURSE core_sources CONFIGURE_DEPENDS "${SQRTVINS_SOURCE}/ov_core/src/*.cpp")
list(FILTER core_sources EXCLUDE REGEX "/test_[^/]*\\.cpp$")
set(srf_sources)
foreach(folder core state update initializer utils)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${SQRTVINS_SOURCE}/ov_srvins/src/${folder}/*.cpp")
    list(APPEND srf_sources ${sources})
endforeach()
add_library(sqrtvins_core STATIC ${core_sources} ${srf_sources})
target_compile_definitions(sqrtvins_core PUBLIC ROS_AVAILABLE=0 ENABLE_ARUCO_TAGS=0 USE_FLOAT=1)
target_include_directories(sqrtvins_core PUBLIC
    ${SQRTVINS_SOURCE}/ov_core/src ${SQRTVINS_SOURCE}/ov_srvins/src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(sqrtvins_core PUBLIC Eigen3::Eigen ${OpenCV_LIBRARIES}
    Boost::filesystem Boost::system Boost::thread Boost::date_time pthread)
add_executable(vslamlab_sqrtvins_stereo_vi runner.cpp)
target_link_libraries(vslamlab_sqrtvins_stereo_vi PRIVATE sqrtvins_core)
