cmake_minimum_required(VERSION 3.16)
project(schurvins_opengv LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
file(GLOB_RECURSE sources "${OPENGV_SOURCE}/src/*.cpp")
add_library(opengv STATIC ${sources})
target_include_directories(opengv PUBLIC ${OPENGV_SOURCE}/include)
target_link_libraries(opengv PUBLIC Eigen3::Eigen)
set_target_properties(opengv PROPERTIES CXX_STANDARD 14)
install(TARGETS opengv ARCHIVE DESTINATION lib)
install(DIRECTORY ${OPENGV_SOURCE}/include/opengv DESTINATION include)
