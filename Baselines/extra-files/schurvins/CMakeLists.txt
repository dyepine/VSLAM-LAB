cmake_minimum_required(VERSION 3.16)
project(vslamlab_schurvins LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system thread)
find_package(OpenMP REQUIRED)
find_package(glog REQUIRED)
find_package(yaml-cpp REQUIRED)
set(S ${SCHURVINS_SOURCE})
set(D ${S}/build/deps)
set(modules svo svo_common svo_direct svo_img_align svo_tracker svo_vio_common
    vikit/vikit_common vikit/vikit_cameras vikit/vikit_solver rpg_common)
set(sources)
foreach(m IN LISTS modules)
 file(GLOB files "${S}/${m}/src/*.cpp")
 list(FILTER files EXCLUDE REGEX "(gpu|test_utils|blender_utils|csv_utils|path_utils)\\.cpp$")
 list(APPEND sources ${files})
endforeach()
file(GLOB fast_sources ${D}/fast_neon/src/*.cpp)
list(FILTER fast_sources EXCLUDE REGEX "(neon|nonmax_5x5)\\.cpp$")
list(APPEND sources ${fast_sources} ${S}/build/generated/svo_factory.cpp)
add_library(schurvins_core STATIC ${sources})
target_include_directories(schurvins_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
 ${S}/build/generated/include ${D}/minkindr/minkindr/include ${D}/fast_neon/include
 ${OPENGV_ROOT}/include ${OpenCV_INCLUDE_DIRS})
foreach(m IN LISTS modules)
 target_include_directories(schurvins_core PUBLIC ${S}/${m}/include)
endforeach()
target_compile_definitions(schurvins_core PUBLIC SVO_USE_OPENGV SVO_DEPTHFILTER_IN_REPROJECTOR)
target_compile_options(schurvins_core PRIVATE -msse3 -mssse3 -Wno-deprecated-declarations)
target_link_libraries(schurvins_core PUBLIC Eigen3::Eigen glog::glog yaml-cpp
 ${OpenCV_LIBS} Boost::filesystem Boost::system Boost::thread ${OPENGV_ROOT}/lib/libopengv.a ${OpenMP_CXX_LIBRARIES} pthread)
add_executable(vslamlab_schurvins_stereo_vi runner.cpp)
target_link_libraries(vslamlab_schurvins_stereo_vi PRIVATE schurvins_core)
